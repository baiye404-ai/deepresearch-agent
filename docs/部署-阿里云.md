
## 重置某个账号的密码

在服务器上执行（用后端自带 venv，避免额外依赖）：

```bash
cd /opt/apps/deepresearch/backend
NEWPW='<新密码>' USER_NAME='<用户名>' .venv/bin/python - <<'PY'
import bcrypt, os, psycopg
from pathlib import Path

env = {}
for line in Path(".env").read_text(encoding="utf-8").splitlines():
    line = line.strip()
    if line and not line.startswith("#") and "=" in line:
        k, v = line.split("=", 1)
        env[k.strip()] = v.strip()
dsn = env["DATABASE_URL"].replace("postgresql+asyncpg://", "postgresql://")

user = os.environ["USER_NAME"]
pw = os.environ["NEWPW"]
hashed = bcrypt.hashpw(pw.encode(), bcrypt.gensalt()).decode()

with psycopg.connect(dsn) as conn, conn.cursor() as cur:
    cur.execute("UPDATE users SET password=%s WHERE username=%s", (hashed, user))
    if cur.rowcount == 0:
        cur.execute("INSERT INTO users (username, password) VALUES (%s, %s)", (user, hashed))
    conn.commit()
    print("done:", user, "rows:", cur.rowcount)
PY
```

改完后无需重启后端（登录时实时查库）。
